Query 7:
SQL:
WITH qualified_users AS (     
    SELECT DISTINCT user_id     
    FROM user_achievements     
    WHERE achievement_type = 'advanced cooking' 
), 
meal_plan_cuisines AS (     
    SELECT mpd.meal_plan_id,            
        COUNT(DISTINCT dc.cuisine_id) AS cuisine_count     
    FROM meal_plan_dishes mpd     
    JOIN dish_cuisine dc ON mpd.dish_id = dc.dish_id     
    JOIN meal_plans m ON mpd.meal_plan_id = m.id     
    JOIN qualified_users qu ON m.user_id = qu.user_id     
    GROUP BY mpd.meal_plan_id 
), 
max_cuisine_count AS (     
    SELECT MAX(cuisine_count) AS max_count     
    FROM meal_plan_cuisines 
) 
SELECT m.id AS meal_plan_id, 
    m.title AS meal_plan_title,     
    GROUP_CONCAT(DISTINCT dr.restriction_name) AS dietary_restrictions,     
    GROUP_CONCAT(DISTINCT ump.preference_type || ': ' || ump.value) AS user_preferences 
FROM meal_plans m 
JOIN qualified_users qu ON m.user_id = qu.user_id 
JOIN meal_plan_cuisines mpc ON m.id = mpc.meal_plan_id 
JOIN max_cuisine_count mcc ON mpc.cuisine_count = mcc.max_count 
LEFT JOIN meal_plan_dishes mpd2 ON m.id = mpd2.meal_plan_id 
LEFT JOIN dish_dietary dd ON mpd2.dish_id = dd.dish_id 
LEFT JOIN dietary_restrictions dr ON dd.dietary_id = dr.id 
LEFT JOIN user_meal_preferences ump ON m.user_id = ump.user_id 
GROUP BY m.id, m.title;

Query 11:
SQL:
SELECT d.bruker_id, COUNT(*) AS total_documents 
FROM (     
    SELECT besv.aktor_id AS bruker_id, COUNT(*) AS doc_count     
    FROM DOKUMENT_BESVARELSE d_b     
    JOIN BESVARELSE besv ON d_b.besvarelse_id = besv.besvarelse_id     
    WHERE d_b.tidspunkt BETWEEN '2023-02-01' AND '2024-08-31'     
    GROUP BY besv.aktor_id     
    HAVING COUNT(*) > 3 
) d 
JOIN RAAD_RATING r ON d.bruker_id = r.bruker_id 
WHERE r.score = 5 
GROUP BY d.bruker_id;

Query 15:
SQL:
WITH moderated_issues AS ( 
    SELECT target_issue_id 
    FROM moderation_log     
    WHERE action_type = 'moderation' 
), 
recent_media_issues AS (     
    SELECT DISTINCT issue_id     
    FROM media     
    WHERE uploaded_at >= DATE('now', '-7 days') 
), 
matched_issues AS (     
    SELECT DISTINCT mi.target_issue_id AS issue_id     
    FROM moderated_issues mi     
    JOIN recent_media_issues rmi ON mi.target_issue_id = rmi.issue_id 
) 
SELECT AVG(tag_count) AS average_tag_count 
FROM (     
    SELECT COUNT(*) AS tag_count     
    FROM issue_tag     
    WHERE issue_id IN (SELECT issue_id FROM matched_issues)     
    GROUP BY issue_id 
);
