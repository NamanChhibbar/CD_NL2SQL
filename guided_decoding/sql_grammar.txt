root ::= ws query ";"

query ::= with_clause? select_stmt

with_clause ::= "WITH" s cte_def (ws "," ws cte_def)* s
cte_def ::= name s "AS" ws "(" ws select_stmt ws ")"

select_stmt ::= "SELECT" s distinct? select_list s "FROM" s from_clause (s where_clause)? (s group_by_clause)? (s having_clause)? (s order_by_clause)? (s limit_clause)?

distinct ::= "DISTINCT" s

select_list ::= select_item (ws "," ws select_item)*
select_item ::= column_ref (s "AS" s alias)?
              | agg_expr (s "AS" s alias)?
              | literal (s "AS" s alias)?
              | "*"

from_clause ::= from_item (s join_clause)*
from_item ::= table_ref (s alias)?
           | "(" ws query ws ")" s alias
join_clause ::= join_type s from_item s "ON" s condition
join_type ::= "JOIN" | "INNER" s "JOIN" | "LEFT" s "JOIN" | "RIGHT" s "JOIN"

where_clause ::= "WHERE" s condition
having_clause ::= "HAVING" s condition

group_by_clause ::= "GROUP" s "BY" s column_ref (ws "," ws column_ref)*
order_by_clause ::= "ORDER" s "BY" s order_item (ws "," ws order_item)*
order_item ::= column_ref (s sort_dir)? | agg_expr (s sort_dir)?
sort_dir ::= "ASC" | "DESC"

limit_clause ::= "LIMIT" s number

table_ref ::= ( __TABLE_REF__ )
column_ref ::= ( __COLUMN_REF__ )

condition ::= condition_atom (s ("AND" | "OR") s condition_atom)*
condition_atom ::= expr s op s expr
                 | expr s "BETWEEN" s expr s "AND" s expr
                 | expr s "IN" ws "(" ws list_values ws ")"
                 | expr s "IN" ws "(" ws query ws ")"
                 | "(" ws condition ws ")"
                 | expr s "LIKE" s string
                 | expr s "IS" s "NULL"
                 | expr s "IS" s "NOT" s "NULL"
                 | "EXISTS" ws "(" ws query ws ")"

list_values ::= literal (ws "," ws literal)*

expr ::= arithmetic_expr
arithmetic_expr ::= term (s add_op s term)*
term ::= factor (s mul_op s factor)*
factor ::= column_ref | literal | agg_expr | "(" ws expr ws ")"

agg_expr ::= agg_func "(" ws agg_arg ws ")"
agg_arg ::= "*" | distinct? column_ref
agg_func ::= "COUNT" | "AVG" | "SUM" | "MAX" | "MIN" | "GROUP_CONCAT"

op ::= ">=" | "<=" | "!=" | "=" | ">" | "<"
add_op ::= "+" | "-"
mul_op ::= "*" | "/"

alias ::= name
literal ::= number | string | "NULL"
name ::= [a-zA-Z_] [a-zA-Z0-9_]*
number ::= "-"? [0-9]+ ("." [0-9]+)?
string ::= "'" ([^'] | "''")* "'"

ws ::= [ \t\n]*
s ::= [ \t\n]+
