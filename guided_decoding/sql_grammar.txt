root ::= ws query ws ";"? ws

query ::= with_clause? select_stmt

with_clause ::= "WITH" ws cte_def (ws "," ws cte_def)* ws
cte_def ::= name ws "AS" ws "(" ws select_stmt ws ")"

select_stmt ::= "SELECT" ws distinct? select_list ws "FROM" ws from_clause (ws where_clause)? (ws group_by_clause)? (ws having_clause)? (ws order_by_clause)?

distinct ::= "DISTINCT" ws

select_list ::= select_item (ws "," ws select_item)*
select_item ::= column_ref (ws "AS" ws alias)?
              | agg_expr (ws "AS" ws alias)?
              | literal (ws "AS" ws alias)?

from_clause ::= table_ref (ws join_clause)*
join_clause ::= join_type ws table_ref ws "ON" ws condition
join_type ::= "JOIN" | "INNER" ws "JOIN" | "LEFT" ws "JOIN" | "RIGHT" ws "JOIN"

where_clause ::= "WHERE" ws condition
having_clause ::= "HAVING" ws condition

group_by_clause ::= "GROUP" ws "BY" ws column_ref (ws "," ws column_ref)*
order_by_clause ::= "ORDER" ws "BY" ws order_item (ws "," ws order_item)*
order_item ::= column_ref (ws sort_dir)? | agg_expr (ws sort_dir)?
sort_dir ::= "ASC" | "DESC"

table_ref ::= ( __TABLE_REF__ )
column_ref ::= ( __COLUMN_REF__ )

condition ::= condition_atom (ws ("AND" | "OR") ws condition_atom)*
condition_atom ::= expr ws op ws expr
                 | expr ws "BETWEEN" ws expr ws "AND" ws expr
                 | expr ws "IN" ws "(" ws list_values ws ")"
                 | "(" ws condition ws ")"

list_values ::= literal (ws "," ws literal)*

expr ::= column_ref | literal

agg_expr ::= agg_func "(" ws agg_arg ws ")"
agg_arg ::= "*" ws | distinct? column_ref
agg_func ::= "COUNT" | "AVG" | "SUM" | "MAX" | "MIN" | "GROUP_CONCAT"

op ::= "=" | "!=" | ">" | "<" | ">=" | "<="

alias ::= name
literal ::= number | string
name ::= [a-zA-Z_] [a-zA-Z0-9_]*
number ::= [0-9]+
string ::= "\"" [^\"]* "\"" | "'" [^']* "'"

ws ::= [ \t\n]*
